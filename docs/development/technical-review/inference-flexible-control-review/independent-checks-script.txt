import json
from pathlib import Path
import numpy as np
from numpy.polynomial.legendre import leggauss
from scipy.linalg import solve_triangular
rng=np.random.default_rng(4905)
errors={key:0.0 for key in ['gaussian_whitening','gaussian_posterior_stationarity','hermite_mass','hermite_stiffness','rotor_trajectory_derivative','rotor_power']}
points,weights=leggauss(4);xi=(points+1)/2;weights=weights/2
for case in range(100):
    raw=rng.normal(size=(3,3));cov=raw@raw.T+np.eye(3);res=rng.normal(size=(5,3));lower=np.linalg.cholesky(cov)
    white=solve_triangular(lower,res.T,lower=True)
    errors['gaussian_whitening']=max(errors['gaussian_whitening'],abs(np.sum(white**2)-np.sum(res*np.linalg.solve(cov,res.T).T)))
    design=rng.normal(size=(3,2));data=rng.normal(size=3);mean=rng.normal(size=2);raw=rng.normal(size=(2,2));prior=raw@raw.T+np.eye(2)
    precision=np.linalg.solve(prior,np.eye(2))+design.T@np.linalg.solve(cov,design)
    post=np.linalg.solve(precision,np.linalg.solve(prior,mean)+design.T@np.linalg.solve(cov,data))
    grad=np.linalg.solve(prior,post-mean)-design.T@np.linalg.solve(cov,data-design@post)
    errors['gaussian_posterior_stationarity']=max(errors['gaussian_posterior_stationarity'],np.max(abs(grad)))
    h=rng.uniform(.2,2);B=rng.uniform(.5,2);mu=rng.uniform(.5,2)
    shape=np.stack([1-3*xi**2+2*xi**3,h*(xi-2*xi**2+xi**3),3*xi**2-2*xi**3,h*(-xi**2+xi**3)],axis=1)
    curvature=np.stack([(-6+12*xi)/h**2,(-4+6*xi)/h,(6-12*xi)/h**2,(-2+6*xi)/h],axis=1)
    integrated_m=mu*h*shape.T@np.diag(weights)@shape
    integrated_k=B*h*curvature.T@np.diag(weights)@curvature
    mass=mu*h/420*np.array([[156,22*h,54,-13*h],[22*h,4*h*h,13*h,-3*h*h],[54,13*h,156,-22*h],[-13*h,-3*h*h,-22*h,4*h*h]])
    stiff=B/h**3*np.array([[12,6*h,-12,6*h],[6*h,4*h*h,-6*h,2*h*h],[-12,-6*h,12,-6*h],[6*h,2*h*h,-6*h,4*h*h]])
    errors['hermite_mass']=max(errors['hermite_mass'],np.max(abs(integrated_m-mass)))
    errors['hermite_stiffness']=max(errors['hermite_stiffness'],np.max(abs(integrated_k-stiff)))
    J,r,F0,tau=rng.uniform(.5,2),rng.uniform(.02,.2),rng.uniform(1,20),rng.uniform(.02,.2)
    u=rng.uniform(0,1,2);a0=rng.uniform(0,1,2);q0,v0=rng.normal(size=2);s=rng.uniform(.01,.5);A=r*F0/J;d=u[0]-u[1];c=a0[0]-a0[1]-d;step=1e-6
    def position(t):return q0+v0*t+A*(d*t*t/2+c*tau*(t-tau*(1-np.exp(-t/tau))))
    def velocity(t):return v0+A*(d*t+c*tau*(1-np.exp(-t/tau)))
    error=max(abs((position(s+step)-position(s-step))/(2*step)-velocity(s)),abs((velocity(s+step)-velocity(s-step))/(2*step)-A*(d+c*np.exp(-s/tau))))
    errors['rotor_trajectory_derivative']=max(errors['rotor_trajectory_derivative'],error)
    q,v=rng.normal(size=2);b,gamma=rng.uniform(0,1,2);a=rng.uniform(0,1,2);torque=r*F0*(a[0]-a[1]);acc=(torque-b*v-gamma*np.sin(q))/J
    def energy(q,v):return J*v*v/2+gamma*(1-np.cos(q))
    finite=(energy(q+step*v,v+step*acc)-energy(q-step*v,v-step*acc))/(2*step)
    errors['rotor_power']=max(errors['rotor_power'],abs(finite-(torque*v-b*v*v)))
record={'seed':4905,'cases_each':100,'maximum_absolute_error':{k:float(v) for k,v in errors.items()},'threshold':1e-6,'scope':'Independent manufactured identities; no empirical validation','scalar_posterior':{'mean':4/3,'variance':1/3},'uniform_beam_frequencies_hz':(np.array([1.8751040687,4.6940911330,7.8547574382])**2/(2*np.pi)).tolist()}
assert max(errors.values())<record['threshold']
Path('reports/technical-review/inference-flexible-control-independent-checks.json').write_bytes((json.dumps(record,indent=2)+'\n').encode())
print(json.dumps(record))
