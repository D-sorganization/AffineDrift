from pathlib import Path
import hashlib,subprocess
bib='articles/The_Geometry_of_Motion/geometry_of_motion.bib'
old=hashlib.sha256(subprocess.check_output(['git','show','HEAD:'+bib])).hexdigest()
new=hashlib.sha256(Path(bib).read_bytes().replace(b'\r\n',b'\n')).hexdigest()
ledger=Path('data/trust/book_publication_audit.json')
s=ledger.read_text(encoding='utf-8')
assert old in s, old
print('Updated book bibliography hashes:',s.count(old))
ledger.write_text(s.replace(old,new),encoding='utf-8')
