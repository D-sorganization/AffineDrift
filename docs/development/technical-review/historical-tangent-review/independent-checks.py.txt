"""Manufactured checks for historical errata; not empirical validation."""
import json
from pathlib import Path
import numpy as np
from scipy.integrate import solve_ivp, quad

rng = np.random.default_rng(4945)
results = []

def record(name, values, tolerance):
    maximum = float(max(values))
    assert maximum < tolerance, (name, maximum, tolerance)
    results.append(dict(name=name, cases=len(values), max_error=maximum, tolerance=tolerance))

adjoint_errors=[]
euler_errors=[]
work_errors=[]
nonlinear_errors=[]
for _ in range(100):
    a,b,u,x0 = rng.uniform(-0.7,0.7,4)
    horizon=float(rng.uniform(0.2,1.0))
    weight=float(rng.uniform(0.2,2.0))
    def cost(command):
        sol=solve_ivp(lambda t,x:a*x+b*command,[0,horizon],[x0],rtol=1e-11,atol=1e-13)
        return .5*sol.y[0,-1]**2+.5*weight*command**2*horizon,sol.y[0,-1]
    value,terminal=cost(u)
    adj=solve_ivp(lambda t,p:-a*p,[horizon,0],[terminal],dense_output=True,rtol=1e-11,atol=1e-13)
    derivative=quad(lambda t:weight*u+b*adj.sol(t)[0],0,horizon,epsabs=1e-12)[0]
    epsilon=1e-5
    numerical=(cost(u+epsilon)[0]-cost(u-epsilon)[0])/(2*epsilon)
    adjoint_errors.append(abs(derivative-numerical))

    rate=float(rng.uniform(.1,1.0))
    exact=np.exp(rate*horizon)
    coarse=abs((1+rate*horizon/200)**200-exact)
    fine=abs((1+rate*horizon/400)**400-exact)
    euler_errors.append(abs(coarse/fine-2.0))

    mass=float(rng.uniform(.1,3.0))
    velocity=rng.normal(size=3)
    force=rng.normal(size=3)
    final_velocity=velocity+horizon*force/mass
    work=quad(lambda t:float(force@(velocity+t*force/mass)),0,horizon,epsabs=1e-12)[0]
    energy=.5*mass*(final_velocity@final_velocity-velocity@velocity)
    impulse=np.array([quad(lambda t:float(f),0,horizon)[0] for f in force])
    work_errors.append(max(abs(work-energy),np.linalg.norm(impulse-mass*(final_velocity-velocity))))

    initial1,initial2=rng.uniform(.02,.15,2)
    def integrate(initial):
        return solve_ivp(lambda t,x:x*x,[0,horizon],[initial],rtol=1e-11,atol=1e-13).y[0,-1]
    combined=integrate(initial1+initial2)
    separate=integrate(initial1)+integrate(initial2)
    assert combined-separate > 0
    exact_combined=(initial1+initial2)/(1-(initial1+initial2)*horizon)
    nonlinear_errors.append(abs(combined-exact_combined))

record('Continuous Adjoint Versus Central Cost Difference',adjoint_errors,1e-7)
record('Euler Global Error Ratio Approaches Two',euler_errors,.02)
record('Constant Force Impulse and Work Balance',work_errors,1e-11)
record('Finite Nonlinear Responses Do Not Superpose',nonlinear_errors,1e-9)
output=Path('reports/technical-review/historical-tangent-independent-checks.json')
output.write_text(json.dumps(dict(seed=4945,checks=results,scope='Manufactured mathematical examples only'),indent=2)+'\n',encoding='utf-8')
print(json.dumps(results,indent=2))
